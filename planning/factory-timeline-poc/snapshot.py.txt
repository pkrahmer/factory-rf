# Embed a snapshot of the files the timeline reads, from a branch of a local clone, into factory-timeline.html.
# Kept as text, so `make lint` does not read it. Usage (from the repository's root):
#   git fetch origin && uv run --no-project python planning/factory-timeline-poc/snapshot.py.txt [origin/main]
import json
import re
import subprocess
import sys
from datetime import datetime, timezone
from pathlib import Path

REF = sys.argv[1] if len(sys.argv) > 1 else "origin/main"
HERE = Path(__file__).parent
WANTED = re.compile(
    r"^(factory/pipeline\.yml|factory/stops\.jsonl|claude/agents/[^/]+\.md|factory/features/[^/]+/feature\.md"
    r"|factory/features/[^/]+/(backlog|in-progress|done)/[^/]+\.(md|costs\.jsonl))$"
)


def git(*args: str) -> str:
    return subprocess.run(["git", *args], check=True, capture_output=True, text=True, encoding="utf-8").stdout


paths = [p for p in git("ls-tree", "-r", "--name-only", REF).splitlines() if WANTED.match(p)]
remote = git("remote", "get-url", "origin").strip()
snapshot = {
    "repo": re.sub(r"^.*github\.com[:/]|\.git$", "", remote),
    "ref": REF.removeprefix("origin/"),
    "sha": git("rev-parse", REF).strip(),
    "taken_at": datetime.now(timezone.utc).isoformat(timespec="seconds").replace("+00:00", "Z"),
    "files": {p: git("show", f"{REF}:{p}") for p in paths},
}
# A work item in progress keeps its journal and cost records on its branch until the merge.
branches = set(git("branch", "-r", "--format=%(refname:short)").split())
for p in paths:
    m = re.match(r"^factory/features/[^/]+/in-progress/(F\d{4}-[SA]\d{4})[^/]*\.(md|costs\.jsonl)$", p)
    if m and f"origin/work-item/{m[1]}" in branches:
        snapshot["files"][p] = git("show", f"origin/work-item/{m[1]}:{p}")
        if p.endswith(".md"):
            costs = p[:-3] + ".costs.jsonl"
            try:
                snapshot["files"][costs] = git("show", f"origin/work-item/{m[1]}:{costs}")
            except subprocess.CalledProcessError:
                pass
data = json.dumps(snapshot, ensure_ascii=False).replace("</", "<\\/")
page = (HERE / "factory-timeline.src.html").read_text(encoding="utf-8").replace("__SNAPSHOT__", data)
(HERE / "factory-timeline.html").write_text(page, encoding="utf-8", newline="\n")
print(f"{len(paths)} files, {len(page) // 1024} KB")
